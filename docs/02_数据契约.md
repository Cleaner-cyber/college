## 1. PlayerState（唯一状态源，Zustand store + 云端/本地存档持久化）

```typescript
interface LogEntry {
  ts: string;                      // ISO 时间
  type: 'level' | 'quick' | 'semester' | 'system';
  text: string;                    // 由引擎用 /content/ui/ui.json 的 log-tmpl 模板生成
}

interface PlayerState {
  version: 2;                      // 存档版本，迁移用
  createdAt: string;               // ISO
  player: {
    name: string;                  // 打字输入①
    majorId: string;               // 来自 majors 库的 id
  };
  flag: {                          // 开局立的四年目标，全点选
    salaryBand: string;            // 如 "20-30w"
    city: string;                  // 如 "一线" | "新一线" | "家乡" | "无所谓"
    workStyle: string;             // 如 "大厂" | "体制内" | "自由职业" | "还没想"
    offTime: string;               // 如 "六点下班" | "不介意加班" | "想自己安排时间"
  };
  axes: {
    academic: number;              // 学术
    portfolio: number;             // 作品
    expression: number;            // 表达
    cash: number;                  // 现金
    energy: number;                // 精力（隐藏轴）
  };
  actionPoints: number;            // 当前学期剩余行动点（只用于选修行动；主线教学关不耗点）
  semester: SemesterId;            // 'prologue' | 'y1s1' | 'y1s1-end'（demo 只有这三个）
  abilities: AbilityId[];          // 已解锁 AI 能力
  archive: ArchiveItem[];          // 档案（驱动简历/结局个性化）
  completedActions: string[];      // 本学期已执行的行动 id（含关卡与速结）
  log: LogEntry[];                 // 全程行动记录（Home「记录」页展示）
}

type AbilityId =
  | 'doc-feeding'     // 投喂长文档（demo：选课关解锁）
  | 'image-gen'       // 生图+迭代（demo：海报关解锁）
  | 'role-play' | 'structured-gen' | 'ai-coding' | 'examiner'
  | 'note-taking' | 'lit-review' | 'data-analysis' | 'multimodal'; // 预留

interface ArchiveItem {
  id: string;                      // 唯一
  levelId: string;                 // 来源关卡/行动
  semester: SemesterId;
  title: string;                   // 展示名，如「社团招新海报 · 双版本」
  resumeLine: string;              // 将来简历关引用的措辞，如「独立完成社团宣传物料设计」
  borrowed: boolean;               // 是否走了[问学长]
  assetRef?: string;               // 关联资产 id（可选）
}
```

## 2. 关卡插件接口

```typescript
interface LevelModule {
  id: string;                              // 与 /content/levels/{id}.json 同名
  Component: React.FC<LevelProps>;
}

interface LevelProps {
  state: Readonly<PlayerState>;
  content: LevelContent;                   // 引擎加载并注入，关卡不自行 import 内容
  onComplete: (result: LevelResult) => void;   // 关卡唯一出口
  onEscape: (result: LevelResult) => void;     // [问学长]出口，result 中条目 borrowed=true
}

interface LevelResult {
  deltas: Partial<PlayerState['axes']>;    // 数值变化
  archiveItems: Omit<ArchiveItem, 'semester'>[];
  abilityUnlocks: AbilityId[];
  checklistScore?: { done: number; total: number };  // 验收清单完成度
}
```

引擎职责：合并 deltas、写入 archive、扣行动点、记录 completedActions、存档、返回行动板。
关卡职责：只渲染流程 + 在结束时调用 onComplete/onEscape。**关卡不直接改 store。**

## 3. 内容 JSON 格式

### 3.1 关卡内容 /content/levels/{levelId}.json
```typescript
interface LevelContent {
  id: string;
  title: string;
  screens: Screen[];
  checklist?: ChecklistItem[];             // 验收清单（任务面板显示）
  presetAssets?: Record<string, string>;   // 逻辑资产id → /public/assets 路径
  copy: Record<string, string>;            // 零散文案键值（按钮、提示等）
}

interface Screen {
  id: string;                              // 如 "S1"
  type: 'dialogue' | 'choice' | 'input' | 'workbench' | 'reveal' | 'deliver';
  speaker?: 'senpai' | 'npc' | 'system';   // dialogue 用
  npcName?: string;                        // 如 "学姐"
  text?: string;                           // 台词/正文，支持 {playerName} {majorName} {clubName} 插值
  choices?: Choice[];                      // choice 用
  inputKey?: string;                       // input 用：写入哪个变量
  placeholder?: string;
  next?: string;                           // 默认跳转屏 id；choice 屏由选项决定
  effects?: ScreenEffects;                 // 该屏触发的表现（见 3.2）
}

interface Choice {
  id: string;
  label: string;
  next: string;
  setVar?: { key: string; value: string }; // 写入关卡局部变量（如风格选择）
  checkItem?: string;                      // 完成某验收项的 id
}

interface ChecklistItem { id: string; label: string; }
```

### 3.2 ScreenEffects（预设演出）
```typescript
interface ScreenEffects {
  typewriter?: boolean;          // 打字机
  fakeLoading?: { ms: number; tips: string[] };  // 假生成等待 + 学长旁批轮播
  showAsset?: string;            // 展示 presetAssets 中的逻辑资产（按局部变量插值，如 "poster-{style}-34"）
  hud?: { hp?: number };         // 心率等 HUD 演出，预留
}
```

## 4. 存档
- **云端（正式）**：Supabase `game_saves` 表（user_id 主键 + state jsonb + RLS 仅本人读写），结构见 /supabase/schema.sql。
- **本地（降级）**：未配置 Supabase 环境变量时用 localStorage，key：`unisim_save_v2`。
- 写入时机：关卡 onComplete/onEscape 与学期流转时立即落盘；速结行动/建档写入防抖 800ms 落盘。
- 读取失败或 version 不匹配 → 清空重开（demo 不做迁移）。
- 引擎只面向 SaveAdapter 接口（src/services/saveAdapter.ts），云端/本地实现可替换。
